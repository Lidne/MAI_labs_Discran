\section{Описание}
Основная идея алгоритма решения задачи состоит в построении суффиксного массива для заданного текста с последующим поиском образцов методом бинарного поиска. Суффиксный массив представляет собой массив индексов всех суффиксов текста, отсортированных в лексикографическом порядке. Это позволяет эффективно находить все вхождения образца в текст за время O(log n) на запрос, где n — длина текста.

Алгоритм состоит из двух основных этапов:
\begin{enumerate}
\item \textbf{Построение суффиксного массива:} Создается массив индексов от 0 до n-1, который сортируется с помощью компаратора, сравнивающего суффиксы текста, начинающиеся с соответствующих позиций.
\item \textbf{Поиск образцов:} Для каждого образца выполняется бинарный поиск в суффиксном массиве для определения диапазона суффиксов, начинающихся с данного образца.
\end{enumerate}

Преимущество данного подхода заключается в том, что при множественных запросах поиска время на построение суффиксного массива (O(n² log n)) окупается эффективностью поиска (O(log n) на запрос) по сравнению с наивным алгоритмом поиска (O(n) на запрос).

\section{Исходный код}
Программа считывает текст с первой строки, строит суффиксный массив, а затем для каждого образца выполняет поиск и выводит позиции вхождений.

\begin{lstlisting}[language=C++, basicstyle=\small\ttfamily]
#include <iostream>
#include <vector>
#include <string>
#include <algorithm>

std::vector<int> buildSA(const std::string &text) {
    int n = text.length();
    std::vector<int> sa(n);
    for (int i = 0; i < n; i++) {
        sa[i] = i;
    }
    auto suffixComparator = [&text](int a, int b) {
     return std::lexicographical_compare(text.begin() + a, text.end(), 
                                        text.begin() + b, text.end()); 
    };
    std::sort(sa.begin(), sa.end(), suffixComparator);
    
    return sa;
}

std::vector<int> searchSA(const std::string &text, const std::string &pattern, 
                         std::vector<int> &sa) {
    int patternLength = pattern.size();
    int arrSize = text.size();
    int l = 0, r = arrSize - 1;
    
    while (l <= r) {
        int center = l + (r - l) / 2;
        if (text.substr(sa[center], patternLength) < pattern) {
            l = center + 1;
        } 
        else {
            r = center - 1;
        }
    }
    
    int firstOccurrence = l;
    r = arrSize - 1;

    while (l <= r) {
        int center = l + (r - l) / 2;
        if (text.substr(sa[center], patternLength) <= pattern) {
            l = center + 1;
        } 
        else {
            r = center - 1;
        }
    }

    int lastOccurrence = r;

    std::vector<int> resultPositions;
    for (int idx = firstOccurrence; idx <= lastOccurrence; idx++) {
        resultPositions.push_back(sa[idx]);
    }

    return resultPositions;
}

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);

    std::string text;
    if (!std::getline(std::cin, text))
        return 0;

    std::vector<int> sa = buildSA(text);

    std::string pattern;
    int count = 1;
    while (std::getline(std::cin, pattern)) {
        std::vector<int> foundPositions = searchSA(text, pattern, sa);
        std::sort(foundPositions.begin(), foundPositions.end());
        if(foundPositions.empty()) {
            count++;
            continue;
        }
        int positionsSize = foundPositions.size();
        std::cout << count << ": ";
        for (int i = 0; i < positionsSize; i++) {
            if(i == positionsSize - 1) {
                std::cout << foundPositions[i] + 1;
            }
            else {
                std::cout << foundPositions[i] + 1 << ", ";
            }
        }
        std::cout << std::endl;
        count++;
    }
    return 0;
}
